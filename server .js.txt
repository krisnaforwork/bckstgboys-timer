const express = require('express');
const http = require('http');
const { Server } = require('socket.io');

const app = express();
const server = http.createServer(app);
const io = new Server(server, {
    cors: { origin: "*" }
});

io.on('connection', (socket) => {
    console.log('Perangkat terhubung:', socket.id);

    // Menerima update dari Controller dan menyiarkannya ke Display
    socket.on('send-state', (data) => {
        socket.broadcast.emit('receive-state', data);
    });

    socket.on('disconnect', () => {
        console.log('Perangkat terputus:', socket.id);
    });
});

const PORT = process.env.PORT || 3000;
server.listen(PORT, () => {
    console.log(`Server WebSocket berjalan di port ${PORT}`);
});
